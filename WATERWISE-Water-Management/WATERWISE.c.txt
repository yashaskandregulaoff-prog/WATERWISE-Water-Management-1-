
#include <stdio.h>

#define ZONES 5
#define DAYS 7

// Function declarations
const char* getZoneName(int zone);

void inputWaterData(int usage[ZONES][DAYS],
                    int dailyTarget[ZONES]);

void displayWaterData(int usage[ZONES][DAYS],
                      int dailyTarget[ZONES]);

void calculateZoneStats(int *zoneUsage, int *total,
                        float *average, int *peakUsage,
                        int *peakDay);

void detectWaterAlerts(int *zoneUsage, int dailyTarget,
                       int *overLimitDays, int *totalExcess,
                       int *spikeDays, int *alertDays);

void generateAuditReport(int usage[ZONES][DAYS],
                         int dailyTarget[ZONES],
                         int excessByZone[ZONES],
                         int overLimitByZone[ZONES],
                         int spikeByZone[ZONES],
                         int alertByZone[ZONES]);

void rankZones(int excessByZone[ZONES]);

void displayConservationActions(int overLimitByZone[ZONES],
                                int spikeByZone[ZONES],
                                int excessByZone[ZONES]);

// Get zone name
const char* getZoneName(int zone)
{
    switch(zone)
    {
        case 0: return "Academic Block";
        case 1: return "Hostel";
        case 2: return "Library";
        case 3: return "Cafeteria";
        case 4: return "Sports Complex";
        default: return "Unknown";
    }
}

// 1. Input water readings
void inputWaterData(int usage[ZONES][DAYS],
                    int dailyTarget[ZONES])
{
    int i, j;

    for(i = 0; i < ZONES; i++)
    {
        printf("\nEnter data for %s\n", getZoneName(i));

        do
        {
            printf("Enter daily target (100-50000): ");
            scanf("%d", &dailyTarget[i]);

            if(dailyTarget[i] < 100 || dailyTarget[i] > 50000)
                printf("Invalid target! Try again.\n");

        } while(dailyTarget[i] < 100 || dailyTarget[i] > 50000);

        for(j = 0; j < DAYS; j++)
        {
            do
            {
                printf("Enter Day %d water usage: ", j + 1);
                scanf("%d", &usage[i][j]);

                if(usage[i][j] < 0 || usage[i][j] > 50000)
                    printf("Invalid reading! Enter 0-50000.\n");

            } while(usage[i][j] < 0 || usage[i][j] > 50000);
        }
    }
}

// 2. Display water data
void displayWaterData(int usage[ZONES][DAYS],
                      int dailyTarget[ZONES])
{
    int i, j;

    printf("\n========== WATER CONSUMPTION DATA ==========\n");

    printf("%-20s", "Zone");

    for(j = 0; j < DAYS; j++)
        printf("Day%-5d", j + 1);

    printf("Target\n");

    for(i = 0; i < ZONES; i++)
    {
        printf("%-20s", getZoneName(i));

        for(j = 0; j < DAYS; j++)
            printf("%-8d", usage[i][j]);

        printf("%d\n", dailyTarget[i]);
    }
}

// 3. Calculate total, average and highest usage
void calculateZoneStats(int *zoneUsage, int *total,
                        float *average, int *peakUsage,
                        int *peakDay)
{
    int i, sum = 0;

    *peakUsage = *zoneUsage;
    *peakDay = 1;

    for(i = 0; i < DAYS; i++)
    {
        sum = sum + *(zoneUsage + i);

        if(*(zoneUsage + i) > *peakUsage)
        {
            *peakUsage = *(zoneUsage + i);
            *peakDay = i + 1;
        }
    }

    *total = sum;
    *average = (float)sum / DAYS;
}

// 4. Detect water alerts
void detectWaterAlerts(int *zoneUsage, int dailyTarget,
                       int *overLimitDays, int *totalExcess,
                       int *spikeDays, int *alertDays)
{
    int i;
    int over, spike;

    *overLimitDays = 0;
    *totalExcess = 0;
    *spikeDays = 0;
    *alertDays = 0;

    for(i = 0; i < DAYS; i++)
    {
        over = 0;
        spike = 0;

        // Check daily target
        if(*(zoneUsage + i) > dailyTarget)
        {
            (*overLimitDays)++;

            *totalExcess = *totalExcess +
                           (*(zoneUsage + i) - dailyTarget);

            over = 1;
        }

        // Check sudden spike from Day 2
        if(i > 0)
        {
            if(*(zoneUsage + i) >
               *(zoneUsage + i - 1) * 1.40)
            {
                (*spikeDays)++;
                spike = 1;
            }
        }

        // Count unique alert days only once
        if(over == 1 || spike == 1)
        {
            (*alertDays)++;
        }
    }
}

// 5. Generate audit report
void generateAuditReport(int usage[ZONES][DAYS],
                         int dailyTarget[ZONES],
                         int excessByZone[ZONES],
                         int overLimitByZone[ZONES],
                         int spikeByZone[ZONES],
                         int alertByZone[ZONES])
{
    int i;
    int total, peakUsage, peakDay;
    int overLimit, excess, spikes, alerts;
    float average;

    printf("\n========== WATER AUDIT REPORT ==========\n");

    for(i = 0; i < ZONES; i++)
    {
        calculateZoneStats(&usage[i][0], &total,
                           &average, &peakUsage, &peakDay);

        detectWaterAlerts(&usage[i][0], dailyTarget[i],
                          &overLimit, &excess,
                          &spikes, &alerts);

        // Store results
        excessByZone[i] = excess;
        overLimitByZone[i] = overLimit;
        spikeByZone[i] = spikes;
        alertByZone[i] = alerts;

        printf("\nZone: %s\n", getZoneName(i));

        printf("Weekly Total       : %d litres\n", total);
        printf("Daily Average      : %.2f litres\n", average);
        printf("Highest Usage      : %d litres\n", peakUsage);
        printf("Highest Usage Day  : Day %d\n", peakDay);

        printf("Over Limit Days    : %d\n", overLimit);
        printf("Total Excess       : %d litres\n", excess);
        printf("Sudden Spike Days  : %d\n", spikes);
        printf("Unique Alert Days  : %d\n", alerts);
    }
}

// 6. Rank zones by excess water
void rankZones(int excessByZone[ZONES])
{
    int i, j, rank;
    int maximum = excessByZone[0];

    printf("\n========== ZONE RANKING ==========\n");

    for(i = 0; i < ZONES; i++)
    {
        if(excessByZone[i] > maximum)
            maximum = excessByZone[i];
    }

    printf("\nZones with greatest excess water:\n");

    for(i = 0; i < ZONES; i++)
    {
        if(excessByZone[i] == maximum)
        {
            printf("%s - %d litres\n",
                   getZoneName(i), maximum);
        }
    }

    printf("\nComplete Ranking:\n");

    for(i = 0; i < ZONES; i++)
    {
        rank = 1;

        for(j = 0; j < ZONES; j++)
        {
            if(excessByZone[j] > excessByZone[i])
                rank++;
        }

        printf("Rank %d : %-20s %d litres\n",
               rank, getZoneName(i), excessByZone[i]);
    }
}

// 7. Display conservation actions
void displayConservationActions(int overLimitByZone[ZONES],
                                int spikeByZone[ZONES],
                                int excessByZone[ZONES])
{
    int i;

    printf("\n========== CONSERVATION ACTIONS ==========\n");

    for(i = 0; i < ZONES; i++)
    {
        printf("\n%s\n", getZoneName(i));

        printf("Total Excess: %d litres\n", excessByZone[i]);

        if(spikeByZone[i] > 0)
        {
            printf("- Check for leaks or unexpected water usage.\n");
        }

        if(overLimitByZone[i] >= 3)
        {
            printf("- Conduct a water audit and reduction plan.\n");
        }
        else if(overLimitByZone[i] == 1 ||
                overLimitByZone[i] == 2)
        {
            printf("- Monitor those days and inspect taps or valves.\n");
        }

        if(overLimitByZone[i] == 0 && spikeByZone[i] == 0)
        {
            printf("- Maintain current good practices.\n");
        }
    }
}

// MAIN FUNCTION
int main()
{
    int usage[ZONES][DAYS];
    int dailyTarget[ZONES];

    int excessByZone[ZONES] = {0};
    int overLimitByZone[ZONES] = {0};
    int spikeByZone[ZONES] = {0};
    int alertByZone[ZONES] = {0};

    int choice;
    int auditDone = 0;

    printf("====================================\n");
    printf("       WATERWISE MANAGEMENT\n");
    printf("       Track. Detect. Conserve.\n");
    printf("====================================\n");

    // Read data only once
    inputWaterData(usage, dailyTarget);

    do
    {
        printf("\n========== MAIN MENU ==========\n");
        printf("1. Display Water Data\n");
        printf("2. Generate Audit Report\n");
        printf("3. Rank Zones by Excess\n");
        printf("4. Conservation Actions\n");
        printf("5. Exit\n");

        printf("Enter your choice: ");
        scanf("%d", &choice);

        switch(choice)
        {
            case 1:
                displayWaterData(usage, dailyTarget);
                break;

            case 2:
                generateAuditReport(usage, dailyTarget,
                                    excessByZone,
                                    overLimitByZone,
                                    spikeByZone,
                                    alertByZone);

                auditDone = 1;
                break;

            case 3:
                if(auditDone == 1)
                {
                    rankZones(excessByZone);
                }
                else
                {
                    printf("Please generate audit report first!\n");
                }
                break;

            case 4:
                if(auditDone == 1)
                {
                    displayConservationActions(overLimitByZone,
                                               spikeByZone,
                                               excessByZone);
                }
                else
                {
                    printf("Please generate audit report first!\n");
                }
                break;

            case 5:
                printf("\nThank you for using WATERWISE!\n");
                printf("Save Water, Save Life!\n");
                break;

            default:
                printf("Invalid choice! Please try again.\n");
        }

    } while(choice != 5);

    return 0;
}
